create or replace view public.user_effective_permissions as
select
  up.id            as user_id,
  p.id             as permission_id,
  p.module,
  p.sub_module,
  p.action,
  p.label,
  coalesce(uperm.is_granted, rp.is_granted, false) as is_granted
from public.user_profiles up
join public.roles r on r.id = up.role_id
left join public.role_permissions rp on rp.role_id = r.id
left join public.permissions p on p.id = rp.permission_id
left join public.user_permissions uperm
  on uperm.user_id = up.id and uperm.permission_id = p.id;

-- is_super_admin
create or replace function public.is_super_admin()
returns boolean
language sql stable security definer as $$
  select exists(
    select 1
    from public.user_profiles up
    join public.roles r on r.id = up.role_id
    where up.id = auth.uid() and r.name = 'super_admin'
  );
$$;

-- is_admin_or_above
create or replace function public.is_admin_or_above()
returns boolean
language sql stable security definer as $$
  select exists(
    select 1
    from public.user_profiles up
    join public.roles r on r.id = up.role_id
    where up.id = auth.uid() and r.name in ('super_admin','admin')
  );
$$;

-- change_user_role
create or replace function public.change_user_role(p_user_id uuid, p_role_name text)
returns void
language plpgsql security definer as $$
declare v_role_id uuid;
begin
  select id into v_role_id from public.roles where name = p_role_name;
  if v_role_id is null then
    raise exception 'Role % not found', p_role_name;
  end if;
  update public.user_profiles
     set role_id = v_role_id, updated_at = now()
   where id = p_user_id;
end;
$$;

-- create_new_user (admin-style creation)
create or replace function public.create_new_user(
  p_email       text,
  p_first_name  text,
  p_last_name   text,
  p_phone       text,
  p_role_name   text
) returns uuid
language plpgsql security definer as $$
declare
  v_user_id  uuid := gen_random_uuid();
  v_role_id  uuid;
begin
  select id into v_role_id from public.roles where name = p_role_name;
  if v_role_id is null then raise exception 'Role % not found', p_role_name; end if;

  insert into auth.users (id, instance_id, email, raw_user_meta_data, aud, role, created_at, updated_at)
  values (v_user_id, '00000000-0000-0000-0000-000000000000', p_email,
          jsonb_build_object('first_name', p_first_name, 'last_name', p_last_name),
          'authenticated', 'authenticated', now(), now())

  values (v_user_id, p_first_name, p_last_name, nullif(p_phone, ''), v_role_id, true)
  on conflict (id) do nothing;

  return v_user_id;
end;
$$;

-- assign_role_permission
create or replace function public.assign_role_permission(
  p_role_name  text,
  p_module     text,
  p_action     text,
  p_sub_module text default null
) returns void
language plpgsql security definer as $$
declare
  v_role_id       uuid;
  v_permission_id uuid;
begin
  select id into v_role_id from public.roles where name = p_role_name;
  if v_role_id is null then raise exception 'Role % not found', p_role_name; end if;

  insert into public.permissions (module, sub_module, action, label)
  values (p_module, p_sub_module, p_action,
          p_module || coalesce('.' || p_sub_module, '') || '.' || p_action)
  on conflict (module, sub_module, action) do nothing;

  select id into v_permission_id
    from public.permissions
   where module = p_module
     and sub_module is not distinct from p_sub_module
     and action = p_action;

  insert into public.role_permissions (role_id, permission_id, is_granted)
  values (v_role_id, v_permission_id, true)
  on conflict (role_id, permission_id) do nothing;
end;
$$;

-- set_user_permission (grant/revoke a single permission for a user)
create or replace function public.set_user_permission(
  p_user_id    uuid,
  p_module     text,
  p_sub_module text,
  p_action     text,
  p_is_granted boolean
) returns void
language plpgsql security definer as $$
declare v_permission_id uuid;
begin
  insert into public.permissions (module, sub_module, action, label)
  values (p_module, p_sub_module, p_action,
          p_module || coalesce('.' || p_sub_module, '') || '.' || p_action)
  on conflict (module, sub_module, action) do nothing;

  select id into v_permission_id
    from public.permissions
   where module = p_module
     and sub_module is not distinct from p_sub_module
     and action = p_action;

  insert into public.user_permissions (user_id, permission_id, is_granted)
  values (p_user_id, v_permission_id, p_is_granted)
  on conflict (user_id, permission_id)
    do update set is_granted = excluded.is_granted;
end;
$$;

-- process_payroll: aligned with the columns your types file expects
create or replace function public.process_payroll(p_period_id uuid)
returns int
language plpgsql security definer as $$
declare
  v_period       public.payroll_periods%rowtype;
  v_count        int := 0;
  v_total_gross  numeric(14,2) := 0;
  v_total_ded    numeric(14,2) := 0;
  v_total_net    numeric(14,2) := 0;
  v_emp          record;
  v_present      int; v_absent int; v_leave int;
  v_ot_hours     numeric(6,2);
  v_basic        numeric(12,2);
  v_ot_pay       numeric(12,2);
  v_gross        numeric(12,2);
  v_tax          numeric(12,2);
  v_ins          numeric(12,2);
  v_total_d      numeric(12,2);
  v_net          numeric(12,2);
begin
  select * into v_period from public.payroll_periods where id = p_period_id;
  if not found then raise exception 'Payroll period not found'; end if;
  if v_period.status <> 'DRAFT' then raise exception 'Period is not DRAFT'; end if;

  delete from public.payroll_records where payroll_period_id = p_period_id;

  for v_emp in
    select e.*, coalesce(s.basic_salary, 0) as s_basic
    from public.employees e
    left join public.salary_structures s on s.employee_id = e.id
    where e.status = 'ACTIVE'
  loop
    v_basic := v_emp.s_basic;

    select
      coalesce(sum(case when status='PRESENT' then 1 else 0 end),0),
      coalesce(sum(case when status='ABSENT'  then 1 else 0 end),0),
      coalesce(sum(case when status='ON_LEAVE' then 1 else 0 end),0),
      coalesce(sum(coalesce(overtime_hours,0)),0)
    into v_present, v_absent, v_leave, v_ot_hours
    from public.attendance
    where employee_id = v_emp.id
      and date between v_period.start_date and v_period.end_date;

    v_ot_pay := round(coalesce(v_basic / 176.0, 0) * 1.5 * v_ot_hours, 2);
    v_gross  := v_basic + v_ot_pay;
    v_tax    := round(v_gross * 0.10, 2);
    v_ins    := round(v_gross * 0.05, 2);
    v_total_d := v_tax + v_ins;
    v_net   := v_gross - v_total_d;

    insert into public.payroll_records (
      payroll_period_id, employee_id,
      present_days, absent_days, leave_days, overtime_hours,
      basic_salary, overtime_pay,
      gross_salary, tax_deduction, insurance_deduction,
      total_deductions, net_salary, status
    ) values (
      p_period_id, v_emp.id,
      v_present, v_absent, v_leave, v_ot_hours,
      v_basic, v_ot_pay,
      v_gross, v_tax, v_ins,
      v_total_d, v_net, 'DRAFT'
    );

    v_total_gross := v_total_gross + v_gross;
    v_total_ded   := v_total_ded + v_total_d;
    v_total_net   := v_total_net + v_net;
    v_count       := v_count + 1;
  end loop;

  update public.payroll_periods
     set total_gross      = v_total_gross,
         total_deductions = v_total_ded,
         total_net        = v_total_net,
         status           = 'PROCESSED',
         processed_at     = now()
   where id = p_period_id;

  return v_count;
end;
$$;

begin
  for t in
    select unnest(array[
      'leave_types','leave_balances','salary_structures',
      'stock_movements','company_settings'
    ])
  loop
    execute format('drop policy if exists "read_all_authenticated" on public.%I', t);

      create policy "read_all_authenticated"
      on public.%I for select
      to authenticated
      using (true)
    $p$, t);

    execute format('drop policy if exists "write_admin" on public.%I', t);
    execute format($p$
      create policy "write_admin"
      on public.%I for all
      to authenticated
      using (public.is_admin_or_above())
      with check (public.is_admin_or_above())
    $p$, t);
  end loop;
end $$;

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
as $$
declare v_default_role uuid;
begin
  select id into v_default_role from public.roles where name = 'employee';
  insert into public.user_profiles (id, first_name, last_name, role_id, is_active)
  values (
    new.id,
    coalesce(new.raw_user_meta_data->>'first_name', split_part(new.email,'@',1)),
    coalesce(new.raw_user_meta_data->>'last_name', ''),
    v_default_role,
    true
  )
  on conflict (id) do nothing;
  return new;
end;
$$;

-- BATCH 20 — PASS 3: DATABASE SCHEMA HARDENING
-- Date applied: 2026-08-12
-- Migration file: supabase/migrations/20260812120000_strengthen_schema.sql
-- Project ref:    pexqlkjlefildktjgrha
-- =====================================================================
-- The previous baseline (20260811220017_remote_schema.sql) had silent
-- data-quality holes — values that should be derived were stored, FKs
-- were missing, RLS was bound to {public} instead of {authenticated},
-- and there was no audit trail. This pass closes all of those without
-- breaking existing data.
--
-- A. GENERATED COLUMN — leave_balances.remaining
--    Before: nullable integer column, could drift out of sync with
--      allocated + carried_over - used.
--    After:  GENERATED ALWAYS AS ((allocated + carried_over) - used)
--      STORED. Application can no longer write a wrong value; the
--      database computes it on every INSERT/UPDATE.
--
-- B. FOREIGN KEY — user_profiles.employee_id → employees.id
--    Before: orphan rows possible (an employee_id pointing at a
--      non-existent employees row).
--    After:  ON DELETE SET NULL (employees.id is uuid, so deleting an
--      employee no longer cascades a profile wipe).
--
-- C. CHECK CONSTRAINTS — 35 added across 13 tables:
--      materials           unit_price >= 0, quantity >= 0
--      stock_items         quantity >= 0, min_quantity >= 0,
--                          min_quantity <= max_quantity (when set)
--      attendance          check_in < check_out (when both set)
--      tasks               progress between 0 and 100, no negative
--                          estimated_hours
--      leave_requests      start_date <= end_date, days_requested > 0
--      leave_balances      allocated/used/carried_over >= 0
--      payroll_records     basic >= 0, allowances/bonus/deductions/tax
--                          >= 0, net = gross - deductions - tax
--      payroll_periods     start_date <= end_date, status in known set
--      projects            start_date <= end_date (when both set),
--                          progress 0..100
--      purchase_orders     total_amount >= 0
--      purchase_order_items quantity > 0, unit_price >= 0,
--                           quantity * unit_price = total_price
--      material_request_items quantity_requested > 0,
--                             quantity_approved <= quantity_requested
--      salary_structures   basic >= 0, allowances/deductions >= 0
--    These now reject bad inserts at the database layer instead of
--    silently corrupting reports.
--
-- D. NOT NULL + DEFAULT — stock_items.quantity
--    Before: nullable — missing quantities produced NULL totals.
--    After:  NOT NULL DEFAULT 0.
--
-- E. NULLABLE COLUMNS — employees
--    department_id, position_id, and phone relaxed to nullable so an
--    employee row can be created at onboarding time before the
--    department/position are finalized. A unique partial index
--    `uniq_user_profiles_employee_id` (WHERE employee_id IS NOT NULL)
--    still prevents two profiles linking to the same employee.
--
-- F. ROW-LEVEL SECURITY — tightened from {public} to {authenticated}
--    70 policies that were bound to {public} were dropped and recreated
--    bound to {authenticated}. Verification query:
--
--      SELECT count(*) FROM pg_policies
--      WHERE schemaname='public' AND 'public' = ANY(roles);
--    → 0 rows (was 70+).
--
--    The 18 policies that were already on {authenticated} were left
--    alone. The 4 mixed {public, authenticated} policies were also
--    narrowed to {authenticated}. Net result: 92 policies on
--    {authenticated}, 0 on {public}.
--
--    Anon users can no longer SELECT/INSERT/UPDATE/DELETE anything in
--    the schema. They must authenticate first (Supabase Auth will issue
--    a JWT containing the auth.uid()).
--
-- G. AUDIT LOGGING — audit_logs table + tg_audit() trigger function
--    New table public.audit_logs(

--      table_name  text    not null,

--      actor_id    uuid    default auth.uid(),

--    purchase_orders, role_permissions, roles, salary_structures,
